cmake_minimum_required(VERSION 3.20)
project(redot_tuber_credential_helper VERSION 1.0.0 LANGUAGES C CXX)

if(WIN32)
  add_executable(redot-tuber-credential-helper windows/main.cpp)
  target_compile_features(redot-tuber-credential-helper PRIVATE cxx_std_17)
  target_compile_definitions(redot-tuber-credential-helper PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(redot-tuber-credential-helper PRIVATE Advapi32)
elseif(UNIX AND NOT APPLE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBSECRET REQUIRED IMPORTED_TARGET libsecret-1)
  add_executable(redot-tuber-credential-helper linux/main.c)
  target_compile_features(redot-tuber-credential-helper PRIVATE c_std_11)
  target_link_libraries(redot-tuber-credential-helper PRIVATE PkgConfig::LIBSECRET)
else()
  message(FATAL_ERROR "Only Windows and Linux x86-64 helpers are in the initial release scope")
endif()

install(TARGETS redot-tuber-credential-helper RUNTIME DESTINATION bin)
